(function () {
  window.__x = {};
  try {
    var c = window.KaiCam && KaiCam.control;
    if (!c) { window.__x = { err: 'no-control', cam: window.__c }; return 'no'; }
    var v = document.getElementById('preview');
    if ('mozSrcObject' in v) v.mozSrcObject = c;
    else v.srcObject = c;
    v.play();
    setTimeout(function () {
      window.__x = {
        readyState: v.readyState,
        videoWidth: v.videoWidth,
        videoHeight: v.videoHeight,
        paused: v.paused,
        currentTime: v.currentTime,
        srcSet: !!(v.mozSrcObject || v.srcObject)
      };
    }, 2500);
    return 'attached';
  } catch (e) { return 'throw ' + e.name + ':' + e.message; }
})()
