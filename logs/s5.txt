(function () {
  try {
    var c = window.KaiCam && KaiCam.control;
    if (!c) return 'no-control';
    var out = {};
    try {
      var r = c.takePicture({ fileFormat: 'jpeg', dateTime: Date.now() }, function (b) {
        window.__photo = 'cb-SUCCESS size=' + (b && b.size);
      }, function () { window.__photo = 'cb-ERR'; });
      out.ret = typeof r;
      if (r && 'onsuccess' in r) {
        r.onsuccess = function () { window.__photo = 'req-SUCCESS size=' + (r.result && r.result.size); };
        r.onerror = function () { window.__photo = 'req-ERR'; };
        out.retIsReq = true;
      }
    } catch (e) { out.tp = 'throw ' + e.name + ':' + e.message; }
    try { c.whiteBalanceMode = 'daylight'; out.wbNow = c.whiteBalanceMode; } catch (e) { out.wb = 'throw'; }
    try { c.isoMode = 'hight-res'; out.isoNow = c.isoMode; } catch (e) { out.iso = 'throw'; }
    try { c.zoom = 2; out.zoomNow = c.zoom; } catch (e) { out.zoom = 'throw'; }
    try { c.exposureCompensation = 1; out.ecNow = c.exposureCompensation; } catch (e) { out.ec = 'throw'; }
    window.__x2 = out;
    return 'fired';
  } catch (e) { return 'throw ' + e.name + ':' + e.message; }
})()
