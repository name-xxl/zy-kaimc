(function () {
  var out = {};
  try {
    var a = window.KaiBt && KaiBt.adapter;
    out.adapter = a ? 'obj' : 'null';
    if (a) {
      try { out.state = a.state; } catch (e) { out.state = 'throw:' + e.name; }
      try { out.enabled = a.enabled; } catch (e) { out.enabled = 'throw:' + e.name; }
      try { out.address = a.address; } catch (e) { out.address = '?'; }
    }
    out.btStatus = window.__bt || '?';
    out.ble = document.getElementById('hud-ble') ? document.getElementById('hud-ble').textContent : '?';
  } catch (e) { out.err = e.name; }
  return JSON.stringify(out);
})()
