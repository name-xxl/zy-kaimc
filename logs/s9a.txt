(function () {
  window.__x = { stage: 'start' };
  try {
    var c = window.KaiCam && KaiCam.control;
    if (!c) { window.__x = { err: 'no-control' }; return 'no'; }
    c.onpicture = function (a) {
      try {
        window.__x.picArg = {
          type: typeof a,
          ctor: (a && a.constructor) ? a.constructor.name : String(a),
          size: (a && a.size !== undefined) ? a.size : 'n/a',
          typeProp: (a && a.type !== undefined) ? a.type : 'n/a',
          keys: a && typeof a === 'object' ? Object.keys(a).slice(0, 8).join(',') : String(a).slice(0, 40)
        };
      } catch (e) { window.__x.picArg = 'inspect-throw ' + e.name; }
    };
    c.takePicture({ fileFormat: 'jpeg', dateTime: Date.now() });
    window.__x.stage = 'fired';
    return 'fired';
  } catch (e) { return 'throw ' + e.name + ':' + e.message; }
})()
